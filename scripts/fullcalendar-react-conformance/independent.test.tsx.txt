import React,{StrictMode,createRef,useState,createContext,useContext} from 'react'
import {render,act,cleanup,fireEvent} from '@testing-library/react'
import Calendar,{formatDate} from '../dist/index'
afterEach(cleanup)
const base={initialDate:'2026-11-03',initialView:'dayGridMonth',timeZone:'UTC'}
function mount(options={}){const ref=createRef<any>(),view=render(<StrictMode><Calendar ref={ref} {...base} {...options}/></StrictMode>);return {api:ref.current.getApi(),view,ref}}
it('does not recurse when datesSet updates the parent',()=>{let calls=0;function App(){const [title,setTitle]=useState('');return <><h1>{title}</h1><Calendar {...base} datesSet={info=>{calls++;setTitle(info.view.title)}}/></>};const view=render(<StrictMode><App/></StrictMode>);expect(view.getByRole('heading',{level:1}).textContent).toContain('November');expect(calls).toBeLessThan(4)})
it('event content inherits React context and latest callbacks',()=>{const C=createContext('');const ref=createRef<any>();function Label(){return <b>{useContext(C)}</b>};const events=[{id:'e',title:'Event',start:'2026-11-03'}];const click=vi.fn();const view=render(<C.Provider value="first"><Calendar {...base} events={events} eventContent={()=><Label/>}/></C.Provider>);expect(view.getByText('first')).toBeTruthy();view.rerender(<C.Provider value="second"><Calendar {...base} ref={ref} events={events} eventContent={()=><Label/>} eventClick={click}/></C.Provider>);fireEvent.click(view.getByText('second'));expect(click.mock.lastCall[0].event.id).toBe('e')})
it('changes events with reversible snapshots and defensive date/property getters',()=>{const change=vi.fn(),{api}=mount({eventChange:change,events:[{id:'e',start:'2026-11-03T10:00:00Z',end:'2026-11-03T11:00:00Z',extendedProps:{x:1}}]});const e=api.getEventById('e');e.start.setUTCFullYear(2000);expect(e.startStr).toContain('2026');act(()=>e.setStart('2026-11-04T10:00:00Z',{maintainDuration:true}));expect(e.endStr).toBe('2026-11-04T11:00:00Z');expect(change.mock.lastCall[0].oldEvent.startStr).toBe('2026-11-03T10:00:00Z');act(()=>change.mock.lastCall[0].revert());expect(e.startStr).toBe('2026-11-03T10:00:00Z');expect(()=>{e.extendedProps.x=2}).toThrow()})
it('clamps month navigation and respects week start/weekends',()=>{const {api,view}=mount({initialDate:'2027-01-31',firstDay:1,weekends:false});act(()=>api.next());expect(api.getDate().toISOString()).toContain('2027-02-28');act(()=>api.changeView('timeGridWeek'));expect(api.view.currentStart.getUTCDay()).toBe(1);expect(view.container.querySelectorAll('.vfc-time-column')).toHaveLength(5)})
it('drops stale async event sources after navigation and unmount',async()=>{const pending:any[]=[],source=(info,ok)=>pending.push(ok);const {api,view}=mount({events:source});act(()=>api.next());await act(async()=>pending.at(-1)([{id:'new',start:'2026-12-03'}]));await act(async()=>pending[0]([{id:'old',start:'2026-11-03'}]));expect(api.getEvents().map(e=>e.id)).toEqual(['new']);view.unmount();pending.forEach(ok=>ok([]));expect(api.getEvents().map(e=>e.id)).toEqual(['new'])})
it('settles rejected sources and exposes loading/failure',async()=>{const failure=vi.fn(),loading=vi.fn();mount({events:()=>Promise.reject(new Error('offline')),eventSourceFailure:failure,loading});await act(async()=>{});expect(failure.mock.lastCall[0].message).toBe('offline');expect(loading.mock.lastCall[0]).toBe(false)})
it('honors selection veto and clears selection',()=>{const select=vi.fn(),unselect=vi.fn(),{api}=mount({select,unselect,selectAllow:info=>info.allDay});act(()=>api.select('2026-11-03T10:00:00Z','2026-11-03T11:00:00Z'));expect(select).not.toHaveBeenCalled();act(()=>api.select('2026-11-03'));expect(select.mock.lastCall[0].endStr).toBe('2026-11-04');act(()=>api.unselect());expect(unselect).toHaveBeenCalledTimes(1)})
it('renders default content on true and does not inject title HTML',()=>{const {view}=mount({events:[{title:'<script>unsafe</script>',start:'2026-11-03'}],eventContent:()=>true});expect(view.getByText('<script>unsafe</script>')).toBeTruthy();expect(view.container.querySelector('script')).toBeNull()})
it('keeps years below 100 and named timezone formatting',()=>{expect(formatDate('0099-02-03',{year:'numeric',timeZone:'UTC'})).toBe('99');expect(formatDate('2020-04-20T15:00:00Z',{hour:'numeric',hour12:false,timeZone:'Africa/Casablanca'})).toBe('15')})
it('rejects unsupported views instead of silently claiming full upstream parity',()=>{const {api}=mount();expect(api.updateSize).toBeUndefined();expect(()=>api.changeView('unsupportedView')).toThrow(/Unsupported calendar view/)})

it('imperative options survive rendering until changed by the caller',()=>{const {api,view}=mount();act(()=>api.setOption('weekends',false));act(()=>api.changeView('timeGridWeek'));expect(view.container.querySelectorAll('.vfc-time-column')).toHaveLength(5);expect(api.getOption('weekends')).toBe(false)})
